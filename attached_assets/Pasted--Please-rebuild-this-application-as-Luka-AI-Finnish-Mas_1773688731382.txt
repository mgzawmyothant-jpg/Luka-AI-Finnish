def handle_search(b):
    if not search_in.value: return
    with out_area:
        clear_output(); b.disabled = True
        print(f"🔍 LUKA is analyzing '{search_in.value}'...")
        res = prof_model.generate_content(f"Analyze the word: {search_in.value}")
        state.last_response = res.text
        clear_output(); display(Markdown(res.text)); b.disabled = False

def handle_ocr(b):
    with out_area:
        clear_output(); uploaded = files.upload()
        if uploaded:
            b.disabled = True; print("🤖 LUKA is processing document (PART 1-4)...")
            img_data = list(uploaded.values())[0]
            res = prof_model.generate_content(["Generate OCR Report in PART 1-4 format from this image:", Image.open(io.BytesIO(img_data))])
            state.last_response = res.text
            clear_output(); display(Markdown(res.text)); b.disabled = False

def start_voice(b):
    output.eval_js('startRecording()')
    rec_start.disabled = True; rec_stop.disabled = False

def stop_voice(b):
    with out_area:
        print("⚡ Processing voice data...")
        audio_data = output.eval_js('stopRecording()')
        if audio_data:
            audio_bytes = base64.decodebytes(audio_data.split(',')[1].encode())
            with open("v.wav", "wb") as f: f.write(audio_bytes)
            trans = stt_model.transcribe("v.wav")
            state.transcripts.append(f"[Voice Memo]: {trans['text']}")
            print(f"✅ Voice Attached: {trans['text']}")
        rec_start.disabled = False; rec_stop.disabled = True

def handle_ask(b):
    with out_area:
        if not chat_in.value and not state.transcripts and not state.files:
            print("⚠️ မေးခွန်း သို့မဟုတ် ဖိုင်တစ်ခုခု ထည့်ပေးပါ။")
            return

        clear_output(); b.disabled = True; b.description = "Thinking..."

        prompt_parts = []
        text_context = chat_in.value
        if state.transcripts:
            text_context += "\n\nVoice Context: " + " ".join(state.transcripts)

        if text_context: prompt_parts.append(text_context)
        else: prompt_parts.append("Please analyze the attached media.")

        for f in state.files: prompt_parts.append(f)

        try:
            res = state.chat_session.send_message(prompt_parts)
            state.last_response = res.text
            clear_output(); display(Markdown(res.text))
        except Exception as e: print(f"Error: {e}")

        b.disabled = False; b.description = "Ask LUKA AI"

def handle_media(b):
    with out_area:
        uploaded = files.upload()
        for fn, content in uploaded.items():
            state.files.append(Image.open(io.BytesIO(content)))
        print(f"✅ Attached {len(uploaded)} file(s).")

def handle_reset_all(b):

    state.chat_session = asst_model.start_chat(history=[])
